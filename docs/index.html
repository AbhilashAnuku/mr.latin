Quick start

Two lines, no build step.

Drop the tag on any page. Mr.Latin finds it, reads the config, and translates everything. Zero runtime dependencies.

<!-- CDN: drop-in, auto-init -->
<script
  src="https://cdn.jsdelivr.net/npm/mr.latin"
  data-mr-latin
  data-languages="en,es,fr,de,ar,hi,te"
></script>
// npm: for apps & bundlers
import MrLatin from "mr.latin";

new MrLatin({
  languages: ["en", "ar", "hi"],
  tone: "warm",
}).start();
